cmake_minimum_required(VERSION 3.18)
project(day06_tiled_matmul VERSION 1.0.0 DESCRIPTION "Day 6: 共享内存优化 — Tiled Matrix Multiply" LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CUDA_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/../common/include)
    set(COMMON_INCLUDE ${CMAKE_CURRENT_SOURCE_DIR}/../common/include)
else()
    set(COMMON_INCLUDE ${CMAKE_CURRENT_SOURCE_DIR}/common/include)
endif()
include_directories(${COMMON_INCLUDE})
if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL "12.0")
    set(CMAKE_CUDA_ARCHITECTURES 50 52 61 70 75 80 86 89 90 100 120)
else()
    set(CMAKE_CUDA_ARCHITECTURES 50 52 61 70 75 80 86)
endif()
add_executable(tiled_matmul src/main.cu)
if(WIN32)
    set_target_properties(tiled_matmul PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
endif()
